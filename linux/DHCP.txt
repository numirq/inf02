INTERFACESv4="enp0s8"  			>>

nano /etc/dhcp/dhcpd.conf		>>
# A slightly different configuration for an internal subnet
subnet [adres sieci] netmask [maska] {
 range [pierwszy adres] [drugi adres];
 option domain-name-servers [adres dns];
 option domain-name "[domena]";
 option routers [adres bramy];
 option broadcast-address [adres broadcast (ostatni)];
 default-lease-time [czas (s)];
 max-lease-time [czas (s)];
}

Przykład


nano /etc/default/isc-dhcp-server  	>>
INTERFACESv4="enp0s8"  			>>

nano /etc/dhcp/dhcpd.conf		>>
# A slightly different configuration for an internal subnet
subnet 192.168.10.0 netmask 255.255.255.0 {
 range 192.168.10.10 192.168.10.20;
 option domain-name-servers 8.8.8.8, 8.8.4.4;
 option domain-name "mzsp.local";
 option routers 192.168.10.1;
 option broadcast-address 192.168.10.255;
 default-lease-time 300000;
 max-lease-time 330000;
}
